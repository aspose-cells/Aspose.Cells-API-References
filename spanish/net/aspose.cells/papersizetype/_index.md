### Ejemplos

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class PaperSizeTypeDemo
    {
        public static void PaperSizeTypeExample()
        {
            // Crear un nuevo libro de trabajo
            Workbook workbook = new Workbook();
            Worksheet worksheet = workbook.Worksheets[0];

            // Acceder al PageSetup de la hoja de cálculo
            PageSetup pageSetup = worksheet.PageSetup;

            // Establece el tamaño de papel a A4
            pageSetup.PaperSize = PaperSizeType.PaperA4;

            // Establece el tamaño de papel a Legal
            pageSetup.PaperSize = PaperSizeType.PaperLegal;

            // Establece el tamaño de papel a Personalizado
            pageSetup.PaperSize = PaperSizeType.Custom;

            // Guarda el libro de trabajo
            workbook.Save("PaperSizeTypeExample.xlsx");

            return;
        }
    }
}
```
