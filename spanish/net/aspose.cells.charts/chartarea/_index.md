### Ejemplos

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using Aspose.Cells.Charts;
    using System;
    using System.Drawing;

    public class ChartAreaDemo
    {
        public static void ChartAreaExample()
        {
            // Instanciando un objeto Workbook
            Workbook workbook = new Workbook();

            // Obteniendo la referencia de la primera hoja de cálculo
            Worksheet worksheet = workbook.Worksheets[0];

            // Añadiendo un valor de muestra a la celda "A1"
            worksheet.Cells["A1"].PutValue(50);

            // Añadiendo un valor de muestra a la celda "A2"
            worksheet.Cells["A2"].PutValue(100);

            // Añadiendo un valor de muestra a la celda "A3"
            worksheet.Cells["A3"].PutValue(150);

            // Añadiendo un valor de muestra a la celda "B1"
            worksheet.Cells["B1"].PutValue(60);

            // Añadiendo un valor de muestra a la celda "B2"
            worksheet.Cells["B2"].PutValue(32);

            // Añadiendo un valor de muestra a la celda "B3"
            worksheet.Cells["B3"].PutValue(50);

            // Agregar un gráfico a la hoja de cálculo
            int chartIndex = worksheet.Charts.Add(ChartType.Column, 5, 0, 15, 5);

            // Accediendo a la instancia del gráfico recién agregado
            Chart chart = worksheet.Charts[chartIndex];

            // Añadiendo NSeries (fuente de datos del gráfico) al gráfico desde la celda "A1" hasta "B3"
            chart.NSeries.Add("A1:B3", true);

            // Obteniendo el área del gráfico
            ChartArea chartArea = chart.ChartArea;

            // Configurando propiedades de ChartArea
            chartArea.X = 10;
            chartArea.Y = 10;
            chartArea.Width = 400;
            chartArea.Height = 300;
            chartArea.IsInnerMode = false;
            chartArea.AutoScaleFont = true;
            chartArea.BackgroundMode = BackgroundMode.Transparent;
            chartArea.IsAutomaticSize = true;
            chartArea.Shadow = true;

            // Estableciendo el color de primer plano del área del gráfico
            chartArea.Area.ForegroundColor = Color.Yellow;

            // Guardando el archivo Excel
            workbook.Save("ChartAreaExample.xlsx");
            workbook.Save("ChartAreaExample.pdf");
        }
    }
}
```
