### أمثلة

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using Aspose.Cells.Charts;
    using System;
    using System.Drawing;

    public class ChartPointDemo
    {
        public static void ChartPointExample()
        {
            // إنشاء كائن Workbook
            Workbook workbook = new Workbook();

            // الحصول على مرجع ورقة العمل الأولى
            Worksheet worksheet = workbook.Worksheets[0];

            // إضافة قيم عينة إلى الخلايا
            worksheet.Cells["A1"].PutValue(50);
            worksheet.Cells["A2"].PutValue(100);
            worksheet.Cells["A3"].PutValue(150);
            worksheet.Cells["B1"].PutValue(60);
            worksheet.Cells["B2"].PutValue(32);
            worksheet.Cells["B3"].PutValue(50);

            // إضافة مخطط إلى ورقة العمل
            int chartIndex = worksheet.Charts.Add(ChartType.PieExploded, 5, 0, 25, 10);

            // الوصول إلى نسخة المخطط المضاف حديثًا
            Chart chart = worksheet.Charts[chartIndex];

            // إضافة NSeries (مصدر بيانات المخطط) إلى المخطط من الخلية "A1" إلى "B3"
            chart.NSeries.Add("A1:B3", true);

            // إظهار تسميات البيانات
            chart.NSeries[0].DataLabels.ShowValue = true;

            // التكرار عبر كل نقطة في السلسلة
            for (int i = 0; i < chart.NSeries[0].Points.Count; i++)
            {
                // الحصول على نقطة البيانات
                ChartPoint point = chart.NSeries[0].Points[i];

                // ضبط انفجار الفطيرة
                point.Explosion = 15;

                // ضبط لون الحدود
                point.Border.Color = Color.Red;

                // تعيين الظل
                point.Shadow = true;

                // تعيين YValue
                point.YValue = 100 + i * 10;

                // تعيين XValue
                point.XValue = "Category " + (i + 1);

                // تعيين IsInSecondaryPlot
                point.IsInSecondaryPlot = false;
            }

            // حفظ ملف Excel
            workbook.Save("ChartPointExample.xlsx");
            workbook.Save("ChartPointExample.pdf");
        }
    }
}
```
