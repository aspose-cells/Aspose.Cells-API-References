```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using Aspose.Cells.Pivot;
    using System;

    public class PivotClassPivotAreaFilterCollectionDemo
    {
        public static void Run()
        {
            // Создать новую книгу
            Workbook workbook = new Workbook();
            Worksheet worksheet = workbook.Worksheets[0];

            // Добавить примерные данные для сводной таблицы
            var cells = worksheet.Cells;
            cells["A1"].Value = "Fruit";
            cells["B1"].Value = "Quantity";
            cells["A2"].Value = "Apple";
            cells["B2"].Value = 10;
            cells["A3"].Value = "Orange";
            cells["B3"].Value = 15;
            cells["A4"].Value = "Banana";
            cells["B4"].Value = 20;
            cells["A5"].Value = "Apple";
            cells["B5"].Value = 5;

            // Создать сводную таблицу
            int index = worksheet.PivotTables.Add("A1:B5", "C3", "PivotTable1");
            PivotTable pivotTable = worksheet.PivotTables[index];
            pivotTable.AddFieldToArea(PivotFieldType.Row, "Fruit");
            pivotTable.AddFieldToArea(PivotFieldType.Data, "Quantity");

            // Создайте PivotAreaFilterCollection.
            PivotAreaFilterCollection filters = new PivotAreaFilterCollection();

            // Создайте и добавьте фильтры (хотя отражение API показывает ограниченную функциональность).
            // Примечание: фактическое создание фильтра потребует более полной информации об API.
            // Это демонстрирует структуру коллекции
            
            // Доступ к элементам (только для чтения)
            if (filters.Count > 0)
            {
                var firstFilter = filters[0];
            }

            // Сохраните книгу
            workbook.Save("PivotAreaFilterCollectionDemo.xlsx");
        }
    }
}
```
