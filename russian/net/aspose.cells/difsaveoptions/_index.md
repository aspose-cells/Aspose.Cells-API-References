### Примеры

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class DifSaveOptionsDemo
    {
        public static void DifSaveOptionsExample()
        {
            // Создайте новый объект Workbook
            Workbook workbook = new Workbook();
            
            // Добавьте некоторые данные в лист для демонстрационных целей
            Worksheet worksheet = workbook.Worksheets[0];
            worksheet.Cells[0, 0].PutValue("Sample Data");

            // Создайте экземпляр DifSaveOptions
            DifSaveOptions saveOptions = new DifSaveOptions
            {
                // Установите свойства согласно спецификации JSON
                ClearData = true,
                CachedFileFolder = @"C:\Temp\CachedFiles",
                ValidateMergedAreas = true,
                MergeAreas = false,
                CreateDirectory = true,
                SortNames = true,
                SortExternalNames = false,
                RefreshChartCache = true,
                UpdateSmartArt = false
            };


            // Сохраните книгу с помощью DifSaveOptions
            workbook.Save("E:\\VSCellsForm\\DifSaveOptionsExample.dif", saveOptions);

            // Выведите в консоль сообщение о завершении процесса
            Console.WriteLine("Workbook saved successfully with DifSaveOptions.");
        }
    }
}
```
