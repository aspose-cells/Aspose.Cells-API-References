### Примеры

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class ValidationCollectionDemo
    {
        public static void ValidationCollectionExample()
        {
            // Создать новую книгу
            Workbook workbook = new Workbook();
            // Получить доступ к первому листу
            Worksheet worksheet = workbook.Worksheets[0];
            // Получите доступ к коллекции проверок листа
            ValidationCollection validations = worksheet.Validations;

            // Определить область ячеек для проверки
            CellArea area = CellArea.CreateCellArea(0, 0, 1, 1);

            // Добавьте проверку в коллекцию
            int validationIndex = validations.Add(area);
            Validation validation = validations[validationIndex];

            // Установить свойства проверки
            validation.Type = ValidationType.List;
            validation.Formula1 = "a,b,c,d";

            // Сохраните книгу
            workbook.Save("ValidationCollectionExample.xlsx");

            return;
        }
    }
}
```
