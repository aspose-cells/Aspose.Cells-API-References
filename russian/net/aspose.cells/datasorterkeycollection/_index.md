### Примеры

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class DataSorterKeyCollectionDemo
    {
        public static void DataSorterKeyCollectionExample()
        {
            // Создать новый объект Workbook
            Workbook workbook = new Workbook();
            Worksheet worksheet = workbook.Worksheets[0];

            // Добавить некоторые образцы данных в лист
            worksheet.Cells["A1"].PutValue("Name");
            worksheet.Cells["B1"].PutValue("Score");
            worksheet.Cells["A2"].PutValue("John");
            worksheet.Cells["B2"].PutValue(85);
            worksheet.Cells["A3"].PutValue("Jane");
            worksheet.Cells["B3"].PutValue(90);
            worksheet.Cells["A4"].PutValue("Doe");
            worksheet.Cells["B4"].PutValue(80);

            // Получите объект DataSorter книги
            DataSorter sorter = workbook.DataSorter;

            // Установите первый порядок для объекта DataSorter
            sorter.Order1 = Aspose.Cells.SortOrder.Descending;
            // Определить первый ключ
            sorter.Key1 = 1; // Sorting by the second column (Score)

            // Создать область ячеек (диапазон)
            CellArea ca = new CellArea
            {
                StartRow = 1,
                StartColumn = 0,
                EndRow = 3,
                EndColumn = 1
            };

            // Отсортируйте данные в указанном диапазоне (A2:B4)
            sorter.Sort(worksheet.Cells, ca);

            // Получите доступ к DataSorterKeyCollection
            DataSorterKeyCollection keys = sorter.Keys;

            // Отобразите количество ключей
            Console.WriteLine("Number of keys: " + keys.Count);

            // Получите доступ к первому ключу и отобразите его свойства
            DataSorterKey key = keys[0];
            Console.WriteLine("Key Index: " + key.Index);
            Console.WriteLine("Key Order: " + key.Order);

            // Сохраните книгу
            workbook.Save("DataSorterKeyCollectionExample.xlsx");
            workbook.Save("DataSorterKeyCollectionExample.pdf");
            return;
        }
    }
}
```
