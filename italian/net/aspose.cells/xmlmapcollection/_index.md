### Esempi

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class XmlMapCollectionDemo
    {
        public static void XmlMapCollectionExample()
        {
            // Crea un nuovo workbook
            Workbook workbook = new Workbook();

            // Accedi a XmlMapCollection dal workbook
            XmlMapCollection xmlMaps = workbook.Worksheets.XmlMaps;

            // Aggiungi un nuovo XmlMap alla raccolta
            int xmlMapIndex = xmlMaps.Add("sample.xml");

            // Accedi al XmlMap appena aggiunto
            XmlMap xmlMap = xmlMaps[xmlMapIndex];

            // Visualizza il conteggio degli XmlMap nella raccolta
            Console.WriteLine("Number of XmlMaps: " + xmlMaps.Count);

            // Imposta la capacità di XmlMapCollection
            xmlMaps.Capacity = 10;

            // Cancella tutti gli XmlMap dalla raccolta
            xmlMaps.Clear();

            // Salva la cartella di lavoro
            workbook.Save("XmlMapCollectionExample.xlsx");

            return;
        }
    }
}
```
