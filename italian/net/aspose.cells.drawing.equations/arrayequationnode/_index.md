### Esempi

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using Aspose.Cells.Drawing;
    using Aspose.Cells.Drawing.Equations;
    using System;

    public class EquationsClassArrayEquationNodeDemo
    {
        public static void Run()
        {
            // Crea un nuovo workbook
            Workbook workbook = new Workbook();
            Worksheet worksheet = workbook.Worksheets[0];

            // Aggiungi una forma di casella di testo per contenere l'equazione
            var textBox = worksheet.Shapes.AddTextBox(0, 0, 0, 0, 100, 100);

            // Crea un nodo di equazione array (deve essere creato con il nodo genitore)
            ArrayEquationNode arrayEquation = (ArrayEquationNode)EquationNode.CreateNode(EquationNodeType.ArrayEquation, workbook, null);

            // Crea un nodo di equazione semplice da aggiungere all'array
            // EquationNode è astratto, quindi dobbiamo creare un tipo specifico
            // Utilizzare il tipo di equazione Testo come l'opzione più semplice
            EquationNode simpleEquation = EquationNode.CreateNode(EquationNodeType.Text, workbook, arrayEquation);

            // Aggiungi l'equazione semplice all'equazione array
            arrayEquation.AddChild(simpleEquation);

            // Imposta l'equazione nella casella di testo
            textBox.Text = arrayEquation.ToMathML();

            // Salva il risultato
            workbook.Save("ArrayEquationNodeDemo.xlsx");
        }
    }
}
```
