### Esempi

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using Aspose.Cells.Drawing;
    using System;
    using System.Drawing;

    public class PictureDemo
    {
        public static void PictureExample()
        {
            // Istanziare un oggetto Workbook
            Workbook workbook = new Workbook();
            Worksheet worksheet = workbook.Worksheets[0];

            // Aggiunta di un'immagine nella posizione di una cella i cui indici di riga e colonna sono 1 nel foglio di lavoro. È la cella "B2".
            int imgIndex = worksheet.Pictures.Add(1, 1, "PictureExample_original.jpg");

            // Ottieni l'oggetto immagine inserito
            Picture pic = worksheet.Pictures[imgIndex];

            // Ottiene l'altezza originale dell'immagine
            int picHeight = pic.OriginalHeight;

            // Ottiene la larghezza originale dell'immagine
            int picWidth = pic.OriginalWidth;

            // Imposta il colore del bordo dell'immagine
            pic.BorderLineColor = Color.Red;

            // Imposta la larghezza del bordo dell'immagine
            pic.BorderWeight = 3;

            // Crea l'oggetto riga della firma
            SignatureLine s = new SignatureLine();
            s.Signer = "Simon Zhao";
            s.Title = "Development Lead";
            s.Email = "Simon.Zhao@aspose.com";

            // Assegna l'oggetto riga della firma all'immagine
            pic.SignatureLine = s;

            // Salva il file Excel
            workbook.Save("PictureExample.xlsx");
        }
    }
}
```
