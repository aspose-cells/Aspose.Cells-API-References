### Exemples

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using Aspose.Cells.Drawing;
    using System;
    using System.Drawing;

    public class PictureDemo
    {
        public static void PictureExample()
        {
            // Instanciation d'un objet Workbook
            Workbook workbook = new Workbook();
            Worksheet worksheet = workbook.Worksheets[0];

            // Ajout d'une image à l'emplacement d'une cellule dont les indices de ligne et de colonne sont 1 dans la feuille de calcul. Il s'agit de la cellule "B2".
            int imgIndex = worksheet.Pictures.Add(1, 1, "PictureExample_original.jpg");

            // Obtenir l'objet image inséré
            Picture pic = worksheet.Pictures[imgIndex];

            // Obtient la hauteur originale de l'image
            int picHeight = pic.OriginalHeight;

            // Obtient la largeur originale de l'image
            int picWidth = pic.OriginalWidth;

            // Définit la couleur de la bordure de l'image
            pic.BorderLineColor = Color.Red;

            // Définit la largeur de la bordure de l'image
            pic.BorderWeight = 3;

            // Créer un objet ligne de signature
            SignatureLine s = new SignatureLine();
            s.Signer = "Simon Zhao";
            s.Title = "Development Lead";
            s.Email = "Simon.Zhao@aspose.com";

            // Attribuer l'objet ligne de signature à l'image
            pic.SignatureLine = s;

            // Enregistrez le fichier Excel
            workbook.Save("PictureExample.xlsx");
        }
    }
}
```
