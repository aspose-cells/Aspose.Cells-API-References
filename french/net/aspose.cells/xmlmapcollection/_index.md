### Exemples

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class XmlMapCollectionDemo
    {
        public static void XmlMapCollectionExample()
        {
            // Créer un nouveau classeur
            Workbook workbook = new Workbook();

            // Accédez à XmlMapCollection depuis le classeur
            XmlMapCollection xmlMaps = workbook.Worksheets.XmlMaps;

            // Ajoutez un nouveau XmlMap à la collection
            int xmlMapIndex = xmlMaps.Add("sample.xml");

            // Accédez au XmlMap nouvellement ajouté
            XmlMap xmlMap = xmlMaps[xmlMapIndex];

            // Affiche le nombre de XmlMaps dans la collection
            Console.WriteLine("Number of XmlMaps: " + xmlMaps.Count);

            // Définissez la capacité de XmlMapCollection
            xmlMaps.Capacity = 10;

            // Efface tous les XmlMaps de la collection
            xmlMaps.Clear();

            // Enregistrez le classeur
            workbook.Save("XmlMapCollectionExample.xlsx");

            return;
        }
    }
}
```
