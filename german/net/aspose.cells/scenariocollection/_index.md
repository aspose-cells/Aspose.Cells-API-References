### Beispiele

```csharp
namespace AsposeCellsExamples
{
    using Aspose.Cells;
    using System;

    public class ScenarioCollectionDemo
    {
        public static void ScenarioCollectionExample()
        {
            // Erstelle ein neues Arbeitsbuch
            Workbook workbook = new Workbook();
            // Greife auf das erste Arbeitsblatt zu
            Worksheet worksheet = workbook.Worksheets[0];

            // Greifen Sie auf die ScenarioCollection des Arbeitsblatts zu
            ScenarioCollection scenarios = worksheet.Scenarios;

            // Fügen Sie ein neues Szenario zur Sammlung hinzu
            int scenarioIndex = scenarios.Add("Scenario1");

            // Greifen Sie auf das neu hinzugefügte Szenario zu
            Scenario scenario = scenarios[scenarioIndex];

            // Setzen Sie einige Eigenschaften für das Szenario
            scenario.Comment = "This is a test scenario.";
            scenario.IsHidden = false;

            // Setze den aktiven Szenario-Index
            scenarios.ActiveIndex = scenarioIndex;

            // Setze den zuletzt ausgewählten Szenario-Index
            scenarios.LastSelected = scenarioIndex;

            // Speichere die Arbeitsmappe
            workbook.Save("ScenarioCollectionExample.xlsx");

            return;
        }
    }
}
```
